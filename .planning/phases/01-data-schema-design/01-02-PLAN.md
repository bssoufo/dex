    1. Create `backend/src/schema/export.py` with two functions:

    ```python
    def export_json_schema(output_path: Path) -> dict:
        """Export SpaModel as JSON Schema file. Returns the schema dict."""
        from .models import SpaModel
        schema = SpaModel.model_json_schema(mode="serialization")
        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_text(json.dumps(schema, indent=2))
        return schema
    ```
