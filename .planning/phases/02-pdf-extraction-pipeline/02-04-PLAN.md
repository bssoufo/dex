       ```python
       import json
       from pathlib import Path
       from src.schema.models import SpaModel

       data_dir = Path("src/data")
       results = {"pass": 0, "fail": 0, "files": []}
       for json_file in data_dir.rglob("*-2026.json"):
           if "-RAW" in json_file.name:
               continue
           try:
               data = json.loads(json_file.read_text())
               SpaModel.model_validate(data)
               has_sources = len(data.get("source_documents", [])) > 0
               has_pages = any(s.get("page_number") for s in data.get("source_documents", []))
               results["pass"] += 1
               results["files"].append({"file": str(json_file), "valid": True, "has_sources": has_sources, "has_pages": has_pages})
           except Exception as e:
               results["fail"] += 1
               results["files"].append({"file": str(json_file), "valid": False, "error": str(e)})

       print(f"\nValidation: {results['pass']} pass, {results['fail']} fail")
       for f in results["files"]:
           status = "PASS" if f["valid"] else "FAIL"
           extra = f" (sources: {f.get('has_sources')}, pages: {f.get('has_pages')})" if f["valid"] else f" ({f.get('error', '')[:80]})"
           print(f"  [{status}] {f['file']}{extra}")
       ```
